static int64_t monotonic_ns(void)
{
    struct timespec t;
    if (clock_gettime(CLOCK_MONOTONIC, &t) != 0) _Exit(2);
    return (int64_t)t.tv_sec * 1000000000LL + t.tv_nsec;
}
static void attach_one_allowed_cpu(void)
{
    cpu_set_t allowed, selected;
    if (sched_getaffinity(0, sizeof(allowed), &allowed) != 0) _Exit(2);
    int chosen = -1;
    for (int i = 0; i < CPU_SETSIZE; ++i) {

        if (CPU_ISSET((size_t)i, &allowed)) { chosen = i; break; }

{

    if (sched_setscheduler(0, SCHED_FIFO, &s) != 0) _Exit(2);
}
static int lock_with_one_millisecond_timeout(void)
{
    const int64_t deadline = monotonic_ns() + 1000000LL;
    const struct timespec abs_deadline = {
        .tv_sec = (time_t)(deadline / 1000000000LL),
        .tv_nsec = (long)(deadline % 1000000000LL)
    };
    if (syscall(SYS_futex, &pi_word, FUTEX_LOCK_PI2, 0, &abs_deadline, NULL, 0) != -1)
        return 0;

}
static void *competing_waiter(void *unused)
{
    (void)unused;
    const int result = lock_with_one_millisecond_timeout();
    if (result != ETIMEDOUT) _Exit(2);
    atomic_store_explicit(&waiter_completed, true, memory_order_release);
    return NULL;
}
static void occupy_cpu_ten_milliseconds(void)
{
    const int64_t start = monotonic_ns();
    while (monotonic_ns() - start < 10000000LL) { }
}

{

2026-11-18 PRIMARY-SOURCE REVIEW OF TWENTY STAFF KVM DESIGN QUESTIONS
The claims are design prompts, not known literal Carl Waldspurger technical-review questions. His 2024-2025 Nutanix-assigned patents corroborate work on virtualization memory reclaim, live migration, and VM I/O, but do not disclose technical-review questions. The Nutanix role itself focuses on KVM, performance, reliability and making hypervisor upgrades faster.

01. Reclaim 32 GB on a host with 20 VMs and CVM.
32 GB is an action target, not proof of available discardable capacity.
32 GB reclaimable needed >= 32 GB requested. If 20 GB can be reclaimed safely, 32 - 20 = 12 GB remains.
Do not assume all CVM pages are pinned because CVM uses VFIO. Determine exact DMA-mapped/pinned bytes. Prefer guest cooperation where possible; quantify guest stalls, disk I/O, swap, zero-page reclamation, guest page cache losses, and storage service headroom. No absolute no-stall promise follows.
Source: https://docs.kernel.org/driver-api/vfio.html
Source: https://docs.kernel.org/core-api/pin_user_pages.html

02. KSM.
Linux KSM has run=0 by default, and requires eligible mappings. "Most production hypervisors disable KSM" is not proven by the default setting. Enable only when repeatable identical memory savings, acceptable scan cost, copy-on-write burden, tenant trust and isolation are shown. The kernel warns that ksm_rmap_items/ksm_merging_pages > 64 for a 64-bit system may make the net KSM profit zero or negative.
Source: https://docs.kernel.org/admin-guide/mm/ksm.html

03. EPT page walks.
Assumptions: 4-level guest paging, 4-level EPT, 4KB guest mappings, no paging-structure caching. Worst-case translation references (excluding final data access): 4*(4+1)+4=24. If all relevant EPT translations terminate in 2MB EPT entries, 4*(3+1)+3=19. Actual costs vary with caching and EPT guest-physical page sizes. 2MB/4KB=512, but splitting a host THP does NOT unconditionally prove split of a given KVM EPT huge mapping.
Source: https://www.intel.com/content/dam/www/public/us/en/documents/research/2010-vol14-iss-3-intel-technology-journal.pdf page 94 of article

04. mmu_notifier.
Correct invariant: a page must not be reused while a stale KVM translation can still reach it. Host memory-management invalidations notify KVM; KVM invalidates relevant mappings and synchronizes translations before freeing/reusing old backing. Exact callbacks/locking depend on Linux revision; do not invent a universal function stack.
Source: https://docs.kernel.org/mm/mmu_notifier.html
Source: https://docs.kernel.org/virt/kvm/locking.html

05. Idle memory tax.
"8 GB touched per second" measures access volume, not distinct touched working set. VM B "has not touched 60 GB" does not prove those bytes are free, nor is it permission to reclaim them without possible stalls. 64+64=128 GB provisioned. If 60 GB truly discardable, 128-60=68 GB residual; if dirty indispensable data, actual reclamation is different. Waldspurger's 2002 policy levies an idle-memory tax with working-set sampling rather than blindly trusting provisioned size.
Source: https://www.waldspurger.org/carl/papers/esx-mem-osdi02.pdf

06. Hot/cold CXL placement.
Correct problem, but "PEBS works as cheap guest hotness" is not a given. OSDI 2024 Memstrata explicitly points out PEBS virtualization compatibility limitations; it uses hardware tiering plus a software cross-VM policy. DAMON clearing accessed bits can trigger expensive TLB shootdowns. Hugepage preservation versus fine-grained placement is an actual measured tradeoff, not a theorem.
Source: https://www.usenix.org/system/files/osdi24-zhong-yuhong.pdf

07. Promotion thrashing.
100-64=36 GB does not fit into local 64 GB. Repeated swaps are possible, not inevitable. Stable set of 64 GB would avoid movement if the valuable pages remain stable. Show displaced-page reuse, bandwidth and useful work saved before asserting thrashing.
Source: https://www.usenix.org/system/files/osdi24-zhong-yuhong.pdf

08. Online MRC.
MRC is an estimated relationship between resource allocation and misses. Under 1% overhead is a target requiring a measurement plan, not a guaranteed property. The 2024 paper reports max 4% of one core and less than 1% of one core per VM for its studied setup; it did not guarantee less than 1% of every guest's throughput.
Source: https://www.usenix.org/system/files/osdi24-zhong-yuhong.pdf

09. CXL bandwidth contention.
180-250 ns is a platform hypothetical, not established universal idle latency. Link saturation creates queues; bound both service rate and arrival bursts. Measure p50, p99, p999, bandwidth demand, fault and migration rates. Choose a policy based on marginal benefit, not idle read latency alone.
Source: https://www.usenix.org/system/files/osdi24-zhong-yuhong.pdf

10. Migration dirty pages.
Write protection and hardware PML find writes; KVM dirty ring transports dirty guest page numbers to QEMU. They are not three interchangeable detector implementations. Hardware logging may still be buffered before userspace reads the ring. For a true exact flow, record the specific kernel version and KVM capabilities.
Source: https://docs.kernel.org/virt/kvm/api.html
Source: https://lkml.rescloud.iu.edu/2412.2/06394.html

11. Convergence.
Precopy cannot converge under a stable dirtying rate greater than sustainable effective transfer rate, after accounting for compression. Example hypothetical: 12GB/s newly dirtied, 10GB/s effective link => at least 2GB/s growth in still-needed bytes. Multifd parallelizes transfer but does not change a fixed link capacity by itself. Compression only helps when data compresses enough after CPU costs. Auto-converge slows guest writes; postcopy runs at destination before all bytes arrive and introduces destination faults and source/destination failure exposure.
Source: https://www.qemu.org/docs/master/devel/migration/postcopy.html
Source: https://www.qemu.org/docs/master/devel/migration/dirty-limit.html

12. VFIO.
"Standard VFIO breaks all live migration" is obsolete as an absolute. QEMU supports live migration of eligible VFIO devices with vendor/device-specific state-save transitions, plus DMA dirtiness accounting. IOMMU tracking alone does not preserve internal device state. QEMU's current VFIO docs state that postcopy migration is not supported for VFIO devices; dirty tracking without IOMMU hardware support has a costly mark-all-dirty fallback.
Source: https://www.qemu.org/docs/master/devel/migration/vfio.html
Source: https://docs.kernel.org/userspace-api/iommufd.html

13. Host upgrade with running guests.
Ordinary kexec alone does not preserve a running QEMU/KVM device and all PCI/IOMMU state. Linux now documents Kexec HandOver (KHO) and Live Update Orchestrator (LUO), with file/device preservation and orchestrated subsystem callbacks. Do not present complete no-drop VM service continuity as already guaranteed on every AHV configuration: version, PCI preservation, KVM state and validation matter. The answer must account for state held by old kernel, host userspace, DMA, IOMMU, interrupts and devices.
Source: https://docs.kernel.org/core-api/kho/index.html
Source: https://docs.kernel.org/core-api/liveupdate.html
Source: https://docs.kernel.org/next/PCI/liveupdate.html

14. CPU-share versus guest lock-holder.
Linux cpu.weight proportionally distributes CPU service among active children; that is NOT useful guest application progress. For active competing weights 200:100, share 2/3 vs 1/3 of the applicable CPU service capacity, not guaranteed 2x operations/s. Guest lock-holder preemption can waste service in a spinning sibling. PLE, guest paravirtual hints, host scheduling and isolation are mitigation opportunities, not guaranteed direct transfer to exact lock owner.
Source: https://docs.kernel.org/admin-guide/cgroup-v2.html

15. CVM latency.
Dedicated CPU cores reduce certain scheduling conflicts but share last-level cache, memory controllers, sockets, device queues and interrupts. It is valid to ask what causes p999 spikes; do not claim all resources interfere in any specific setup without evidence. Check remote-NUMA load, LLC misses, device queue stalls and softirq/IRQ placement.
Source: https://www.nutanixbible.com/5b-book-of-ahv-how-it-works.html

16. vNUMA drift.
Guest-visible NUMA topology can become less faithful to physical backing after host movement or migration. This may reduce locality without violating guest memory correctness. Challenge exact guest policy and host movement, then measure local-vs-remote access and actual application latency; pinning or NUMA-balanced remapping has costs.
Source: https://docs.kernel.org/admin-guide/mm/multigen_lru.html

17. Nutanix AHV storage.
Do not force one unconditional path. Nutanix documents classic: guest SCSI -> virtio-scsi -> QEMU main loop -> libiscsi -> CVM Stargate. It also documents Frodo: guest SCSI -> virtio-scsi -> Frodo vhost-user-scsi -> customized libiscsi -> network -> CVM Stargate. Frodo was described as default for VMs started after AOS 5.5.X. The question assumes Frodo and must state version/configuration. Poll versus interrupt choice requires actual queue latency, cycles per I/O and CPU utilization.
Source: https://www.nutanixbible.com/5b-book-of-ahv-how-it-works.html

18. Profiler overhead.
An observed 8% throughput regression is a customer claim, not yet a causal attribution. Compare identical workloads with profiler off/on using random order and repeated samples; inspect event rate, time in the profiler, LLC/TLB side effects, and distribution of rare 5ms stalls. "Under 1%" is a target, not proof.
Source: https://docs.kernel.org/trace/index.html

19. SMT core scheduling.
Kernel core scheduling uses trust-group cookies. "Only same-VM siblings may co-run" is one possible grouping policy, not default. The kernel documents MDS/L1TF gaps during kernel contexts and brief untrusted simultaneous windows; core scheduling is not a complete security guarantee. Any throughput cost depends on occupancy and scheduling constraints, not one fixed percentage.
Source: https://docs.kernel.org/admin-guide/hw-vuln/core-scheduling.html
Source: https://docs.kernel.org/admin-guide/hw-vuln/l1tf.html

A request to name personally modified kernel source has no externally checkable one-size-fits-all answer. A credible candidate must show an actual change, reason, original failure, invariant, source diff, observed before-and-after and reasons to keep the existing boundary. #include <stddef.h>
#include <stdint.h>

enum { MAX_CLIENTS = 101, MAX_TICKS = 200 };

typedef struct {
    uint32_t tickets;
    uint32_t stride;
    uint32_t pass;
    uint32_t grants;
} Client;

typedef struct {
    Client clients[MAX_CLIENTS];
    uint32_t history[MAX_TICKS];
    uint32_t count;
    uint32_t tick_index;
    uint32_t stride_unit;
} Scheduler;

typedef struct {
    uint32_t paper_sequence[6];
    uint32_t after_100[MAX_CLIENTS];
    uint32_t after_200[MAX_CLIENTS];
    uint32_t baseline_ok;
    uint32_t stress_ok;
} Evidence;

static Scheduler g_scheduler;
static Evidence g_evidence;
static uint32_t g_active_case;

/* HISTORICAL_LIMIT: basic stride cannot bound global short-window error by 1. */

__attribute__((noinline)) static uint32_t compare_pass(uint32_t left, uint32_t right)
{
    uint32_t left_pass = g_scheduler.clients[left].pass;
    uint32_t right_pass = g_scheduler.clients[right].pass;
    if (left_pass < right_pass) {
        return left;
    }
    if (right_pass < left_pass) {
        return right;
    }
    return left < right ? left : right;
}

__attribute__((noinline)) static uint32_t select_better(uint32_t best, uint32_t challenger)
{
    return compare_pass(best, challenger);
}

__attribute__((noinline)) static uint32_t scan_ready(void)
{
    uint32_t best = 0;
    for (uint32_t i = 1; i < g_scheduler.count; ++i) {
        best = select_better(best, i);
    }
    return best;
}

__attribute__((noinline)) static uint32_t queue_remove_min(void)
{
    /* Scan substitutes for the original paper's abstract priority queue. */
    return scan_ready();
}

__attribute__((noinline)) static void allocate_quantum(void)
{
    uint32_t winner = queue_remove_min();
    g_scheduler.history[g_scheduler.tick_index] = winner;
    ++g_scheduler.clients[winner].grants;
    g_scheduler.clients[winner].pass += g_scheduler.clients[winner].stride;
}

__attribute__((noinline)) static void tick_once(uint32_t index)
{
    g_scheduler.tick_index = index;
    allocate_quantum();
}

__attribute__((noinline)) static void drive_burst(uint32_t first, uint32_t last)
{
    for (uint32_t i = first; i < last; ++i) {
        tick_once(i);
    }
}

__attribute__((noinline)) static void run_phase(uint32_t first, uint32_t last)
{
    drive_burst(first, last);
}

static void initialise(uint32_t count, uint32_t stride_unit)
{
    g_scheduler.count = count;
    g_scheduler.stride_unit = stride_unit;
    g_scheduler.tick_index = 0;
    for (uint32_t i = 0; i < count; ++i) {
        g_scheduler.clients[i].tickets = 1;
        g_scheduler.clients[i].stride = stride_unit;
        g_scheduler.clients[i].pass = stride_unit;
        g_scheduler.clients[i].grants = 0;
    }
}


/* Original grant counts are preserved; compare real prefixes to exact shares. */
__attribute__((noinline)) static int64_t signed_gap(uint32_t slots, uint32_t client)
{
    uint64_t total = 0, received = 0;
    for (uint32_t i = 0; i < g_scheduler.count; ++i) total += g_scheduler.clients[i].tickets;
    for (uint32_t i = 0; i < slots; ++i) received += g_scheduler.history[i] == client;
    return (int64_t)(received * total) - (int64_t)((uint64_t)slots * g_scheduler.clients[client].tickets);
}

/* The first completed slot whose global share error exceeds one grant. */
__attribute__((noinline)) static uint32_t first_global_violation(uint32_t slots, uint32_t client)
{
    int64_t total = 0;
    for (uint32_t i = 0; i < g_scheduler.count; ++i) total += g_scheduler.clients[i].tickets;
    for (uint32_t n = 1; n <= slots; ++n) {
        int64_t gap = signed_gap(n, client);
        if (gap > total || gap < -total) return n;
    }
    return 0;
}

static int run_baseline(void)
{
    initialise(3, 6);
    g_active_case = 3;
    g_scheduler.clients[0].tickets = 3;
    g_scheduler.clients[1].tickets = 2;
    g_scheduler.clients[2].tickets = 1;
    for (uint32_t i = 0; i < 3; ++i) {
        g_scheduler.clients[i].stride = g_scheduler.stride_unit / g_scheduler.clients[i].tickets;
        g_scheduler.clients[i].pass = g_scheduler.clients[i].stride;
    }
    run_phase(0, 6);
    static const uint32_t expected[6] = {0, 1, 0, 0, 1, 2};
    for (uint32_t i = 0; i < 6; ++i) {
        g_evidence.paper_sequence[i] = g_scheduler.history[i];
        if (g_scheduler.history[i] != expected[i]) {
            return 1;
        }
    }
    g_evidence.baseline_ok = 1;
    return 0;
}

__attribute__((noinline)) static int run_stress(void)
{
    initialise(101, 100);
    g_active_case = 101;
    g_scheduler.clients[0].tickets = 100;
    g_scheduler.clients[0].stride = g_scheduler.stride_unit / 100;
    g_scheduler.clients[0].pass = g_scheduler.clients[0].stride;
    run_phase(0, 100);
    for (uint32_t i = 0; i < MAX_CLIENTS; ++i) {
        g_evidence.after_100[i] = g_scheduler.clients[i].grants;
    }
    if (g_evidence.after_100[0] != 100) {
        return 2;
    }
    for (uint32_t i = 1; i < MAX_CLIENTS; ++i) {
        if (g_evidence.after_100[i] != 0) {
            return 3;
        }
    }
    run_phase(100, 200);
    for (uint32_t i = 0; i < MAX_CLIENTS; ++i) {
        g_evidence.after_200[i] = g_scheduler.clients[i].grants;
    }
    if (g_evidence.after_200[0] != 100) {
        return 4;
    }
    for (uint32_t i = 1; i < MAX_CLIENTS; ++i) {
        if (g_evidence.after_200[i] != 1) {
            return 5;
        }

    if (signed_gap(100, 0) != 10000 || signed_gap(100, 1) != -100 ||
        signed_gap(200, 0) != 0 || first_global_violation(100, 0) != 3) {

    }
    g_evidence.stress_ok = 1;
    return 0;

    if (run_baseline() != 0) {
        return 10;
    }
    return run_stress();

    return run_all();

WHY THE TOPICS ARE CONNECTED

For CPU time, the allocator sees tickets, stride and pass values. In the documented 101-client case, the next-recipient rule can give program 0 all 100 early grants. The long-run share is correct, but short-window global error is 50. The distinction is the competition set and time interval, not a missing arithmetic operation.

For virtualized memory, an ESX host controls machine pages, while a guest OS has information about its own replacement choices. Ballooning requests a guest-controlled reclaim decision; forced host paging cannot use exactly the same guest information. Transparent page sharing takes the opposite information problem: separate guest processes can contain identical page bytes without having a known shared page-cache origin, so discovering byte equality is an additional mechanism.

For Linux KSM, equality is not enough for safety. The November 2008 discussion and April 2009 review involve page-table replacement, COW behavior, and shared writable mappings. Study the original patch assumptions before changing present-day mm/ code.

For a balloon driver, an otherwise valid request can become incorrect when the host target changes during the request. A 2010 Linux driver patch bounded a refused-page list. A bound controls memory use; it does not prove the host will eventually accept an allocation.

These are related resource-control design constraints, not proof of one common original implementation. Work by later Linux maintainers is not attributed to Waldspurger without a primary authorship record.

The core causal chain and the study questions supported by these sources

CPU time allocation: A process claims a fraction of processing capacity. Lottery scheduling allocates service probabilistically according to tickets. Stride scheduling accounts for each service event using a deterministic pass value, which reduces allocation variance. The design question is how a policy stays fair when tasks sleep, awaken, join, depart, or change weight.

Virtual-machine memory allocation: The host owns physical machine frames, but the guest chooses which guest pages are least useful. A host-only page eviction decision can harm the guest because the hypervisor lacks complete knowledge of its replacement policy. Ballooning transfers the reclamation decision to the guest by causing it to free its own pages. The design question is what happens if the host changes its request during balloon inflation, and what state the driver must bound or reset.

Transparent page sharing: Two VMs can independently hold identical contents. Their different origins mean ordinary shared-file assumptions may not discover that duplication. Deduplication compares page contents and safely replaces mappings with a shared read-only backing page, with copying required upon writes. The design question is how to update all mappings and reference counts while preserving permissions and handling concurrent writes, migration, and invalidation.

Memory resource controls: An allocation policy must balance actual pressure, idle memory, memory access measurement, and guest isolation. If the hypervisor reclaims too early, paging overhead rises. If it waits too long, another VM cannot obtain needed frames. Waldspurger's ESX paper should be used to derive the policy, not to copy its historical numeric thresholds as modern kernel defaults.

Chronological reading sequence for direct source verification

Read the 1994 lottery paper, then the 1995 MIT PhD thesis. Work through the 1996 UW scheduling problem to test the scheduler's goal against different workloads. Read the 2001 MIT scheduling and page-replacement notes. Implement the 2002 to 2003 MIT OS exercises for page tables and preemption. Read the 2002 VMware ESX original paper and the 2005 UT Austin and 2006 Berkeley graduate-class assignments of that paper. Compare the 2008 Stanford virtual-memory project with the November 2008 KSM design thread. Inspect the April 2009 KSM correction and May 2010 balloon driver patch. Use the 2009 to 2011 MIT 6.828 labs to see page tables, scheduling, and VMs in the same codebase. Reserve later CXL work for modern resource-allocation tradeoffs.

Primary-source causal reconstruction and local debugger evidence
18 November 2026

The governing issue is not whether a resource is divided in a ratio.
The issue is what the ratio guarantees, over which group, in which interval,
and which component knows enough to make the next allocation.

A short-window failure appears in Waldspurger and Weihl's June 1995 MIT
memo, and in Waldspurger's September 1995 doctoral thesis.

100 slots are offered to 101 always-runnable programs.
Program 0 holds 100 tickets. Programs 1 through 100 hold 1 ticket each.
Total tickets = 100 + 100 * 1 = 200.
Program 0 is entitled to 100 * 100 / 200 = 50 of the first 100 slots.
Basic stride scheduling can give program 0 all 100 slots initially.
Observed count in the local reproduction = 100.
Absolute short-window error = 100 - 50 = 50 slots.

A pairwise calculation uses a different denominator.
Compare program 0 with program 1, excluding the other 99 programs.
The pair received 100 slots; program 0's share inside the pair is 100/101.
Pairwise ideal for program 0 = 100 * 100 / 101 = 99.00990099.
Pairwise error = 100 - 99.00990099 = 0.99009901 slot.
The pairwise error is below 1 although the global error is 50.
No contradiction: the pairwise and global guarantees measure different sets.

Waldspurger's original algorithm stores one ticket count, stride and pass
value per program. A larger ticket count gives a smaller stride.
The next recipient has the smallest pass value.
After a grant, pass increases by the recipient's stride.
Ties are broken by a fixed order. The original paper permits this.

The code below tests the original paper's 3:2:1 example, then its 101-client
pathological case. Client 0 wins deterministic ties.

How the next result follows from the entry point

main -> run_all -> run_stress -> run_phase -> drive_burst -> tick_once
-> allocate_quantum -> queue_remove_min -> scan_ready -> select_better
-> compare_pass.

The entry sets 101 clients. One receives 100 tickets, and 100 receive one.
For client 0, stride = 100 / 100 = 1; initial pass = 1.
For client 1, stride = 100 / 1 = 100; initial pass = 100.
The next-pass comparison selects client 0 while its pass is below 100.
At pass = 100, equal values select client 0 because 0 precedes 1.
After the grant, client 0's pass becomes 101.
Every smaller client still has pass 100, so all 100 become eligible
before client 0's next turn. The next 100 slots go to those clients.
After 200 slots: client 0 = 100; each other client = 1.
The long-run share is correct although the short-window error was large.

What Waldspurger's other papers add to the same question

1989 MIT master's thesis:
A Distributed Computational Economy for Utilizing Idle Resources.
Resource ownership and allocation are negotiated across machines.
Use the thesis for historical origins, not as evidence of later KVM code.

1994 OSDI paper:
Lottery Scheduling: Flexible Proportional-Share Resource Management.
Random selection makes dynamic resource changes easier, but service
rates fluctuate across short windows. The paper reports a Mach 3.0 prototype.

1995 MIT memo and thesis:
Stride scheduling replaces random choice with a stored next-turn value.
The benefit is deterministic pairwise accuracy.
The failure is that pairwise accuracy alone does not bound global error.
Hierarchical stride grouping reduces worst-case absolute error from
an amount proportional to client count to an amount growing with its log.
This is an asymptotic mathematical result under the paper's model, not a
claim of a tested modern Linux scheduler implementation.

2002 ESX memory paper:
A virtual machine monitor can reclaim memory, but it does not possess
all the guest OS's page-replacement knowledge. Forced host paging can
cause double paging. The balloon driver instead requests pages from
inside the guest, causing the guest to use its own replacement policy.
The guest supplies page numbers to the hypervisor for reclamation.
The hypervisor retains responsibility for machine-page mappings.

The paper also shows why pure share ratios fail for idle memory.
One VM can hold large memory allocations while touching few pages.
The default idle tax in the paper is 75%, giving idle page cost
1 / (1 - 0.75) = 4 in its adjusted shares-per-page metric.

For an illustrative 100-share VM allocated 100 pages with 25 active:
Active fraction = 25 / 100 = 0.25.
Adjusted equivalent page count = 100 * (0.25 + 4 * 0.75) = 325.
Adjusted shares per page = 100 / 325 = 0.3076923077.
With all 100 pages active, adjusted count = 100, ratio = 1.
The ratio controls reclamation preference; it is not a command to
remove 75 pages immediately.

The original 2002 sampling method selected 100 pages per 30-second period.
It invalidated cached guest-page translations, counted touches, then
estimated the active fraction by touched / sampled. Three smoothed
estimates supported fast expansion but slower contraction of the estimate.
An estimate does not identify every cold page with certainty.

2010 LKML patch, not authored by Waldspurger:
The VMware balloon driver could keep collecting refused pages while
its hypervisor's target changed. An unbounded retry does not become
safe merely because each individual page allocation is legal.
The May 2010 patch limited accumulated refused pages to 16 per cycle.
This is a documented historical Linux driver change. No VMware guest
or hypervisor was run in this local reproduction.

2012 VMware distributed resource management paper:
VM placement adds another decision layer. A VM that fits on an individual
host may still create imbalance across a cluster. CPU, memory, power and
placement constraints must be considered together.

The likely methodological pattern, inferred from publications rather
than personal knowledge of Waldspurger's private reasoning, is:
Choose an explicit service objective.
State which programs are competing and over what time window.
Construct the smallest counterexample that defeats a weak guarantee.
Add a mechanism that uses the information available at its own layer.
Check both short-term damage and long-term resource utilization.

Errors in reasoning to watch for

Saying "proportional" without defining the time interval.
Using a pairwise theorem to claim an all-client bound.
Mistaking a 14-frame call stack for proof that the historical kernel ran.
Mistaking an idle-memory weighting formula for an immediate page count.
Assuming host and guest know the same page-access information.
Treating the documented 2010 balloon patch as Waldspurger's authorship.
Treating a known algorithmic limitation as a spelling or compiler error.
Inferring an author's private thoughts from papers rather than deriving
claims from the published formulas and measurements.

BEGIN_C_SOURCE
#include <stddef.h>
#include <stdint.h>

enum { MAX_CLIENTS = 101, MAX_TICKS = 200 };

typedef struct {
    uint32_t tickets;
    uint32_t stride;
    uint32_t pass;
    uint32_t grants;
} Client;

typedef struct {
    Client clients[MAX_CLIENTS];
    uint32_t history[MAX_TICKS];
    uint32_t count;
    uint32_t tick_index;
    uint32_t stride_unit;
} Scheduler;

typedef struct {
    uint32_t paper_sequence[6];
    uint32_t after_100[MAX_CLIENTS];
    uint32_t after_200[MAX_CLIENTS];
    uint32_t baseline_ok;
    uint32_t stress_ok;
} Evidence;

static Scheduler g_scheduler;
static Evidence g_evidence;
static uint32_t g_active_case;

/* USER_SOURCE_ERRORS: none provided; no user source was edited. */
/* HISTORICAL_LIMIT: basic stride cannot bound global short-window error by 1. */

__attribute__((noinline)) static uint32_t compare_pass(uint32_t left, uint32_t right)
{
    uint32_t left_pass = g_scheduler.clients[left].pass;
    uint32_t right_pass = g_scheduler.clients[right].pass;
    if (left_pass < right_pass) {
        return left;
    }
    if (right_pass < left_pass) {
        return right;
    }
    return left < right ? left : right;
}

__attribute__((noinline)) static uint32_t select_better(uint32_t best, uint32_t challenger)
{
    return compare_pass(best, challenger);
}

__attribute__((noinline)) static uint32_t scan_ready(void)
{
    uint32_t best = 0;
    for (uint32_t i = 1; i < g_scheduler.count; ++i) {
        best = select_better(best, i);
    }
    return best;
}

__attribute__((noinline)) static uint32_t queue_remove_min(void)
{
    /* Scan substitutes for the original paper's abstract priority queue. */
    return scan_ready();
}

__attribute__((noinline)) static void allocate_quantum(void)
{
    uint32_t winner = queue_remove_min();
    g_scheduler.history[g_scheduler.tick_index] = winner;
    ++g_scheduler.clients[winner].grants;
    g_scheduler.clients[winner].pass += g_scheduler.clients[winner].stride;
}

__attribute__((noinline)) static void tick_once(uint32_t index)
{
    g_scheduler.tick_index = index;
    allocate_quantum();
}

__attribute__((noinline)) static void drive_burst(uint32_t first, uint32_t last)
{
    for (uint32_t i = first; i < last; ++i) {
        tick_once(i);
    }
}

__attribute__((noinline)) static void run_phase(uint32_t first, uint32_t last)
{
    drive_burst(first, last);
}

static void initialise(uint32_t count, uint32_t stride_unit)
{
    g_scheduler.count = count;
    g_scheduler.stride_unit = stride_unit;
    g_scheduler.tick_index = 0;
    for (uint32_t i = 0; i < count; ++i) {
        g_scheduler.clients[i].tickets = 1;
        g_scheduler.clients[i].stride = stride_unit;
        g_scheduler.clients[i].pass = stride_unit;
        g_scheduler.clients[i].grants = 0;
    }
}

static int run_baseline(void)
{
    initialise(3, 6);
    g_active_case = 3;
    g_scheduler.clients[0].tickets = 3;
    g_scheduler.clients[1].tickets = 2;
    g_scheduler.clients[2].tickets = 1;
    for (uint32_t i = 0; i < 3; ++i) {
        g_scheduler.clients[i].stride = g_scheduler.stride_unit / g_scheduler.clients[i].tickets;
        g_scheduler.clients[i].pass = g_scheduler.clients[i].stride;
    }
    run_phase(0, 6);
    static const uint32_t expected[6] = {0, 1, 0, 0, 1, 2};
    for (uint32_t i = 0; i < 6; ++i) {
        g_evidence.paper_sequence[i] = g_scheduler.history[i];
        if (g_scheduler.history[i] != expected[i]) {
            return 1;
        }
    }
    g_evidence.baseline_ok = 1;
    return 0;
}

__attribute__((noinline)) static int run_stress(void)
{
    initialise(101, 100);
    g_active_case = 101;
    g_scheduler.clients[0].tickets = 100;
    g_scheduler.clients[0].stride = g_scheduler.stride_unit / 100;
    g_scheduler.clients[0].pass = g_scheduler.clients[0].stride;
    run_phase(0, 100);
    for (uint32_t i = 0; i < MAX_CLIENTS; ++i) {
        g_evidence.after_100[i] = g_scheduler.clients[i].grants;
    }
    if (g_evidence.after_100[0] != 100) {
        return 2;
    }
    for (uint32_t i = 1; i < MAX_CLIENTS; ++i) {
        if (g_evidence.after_100[i] != 0) {
            return 3;
        }
    }
    run_phase(100, 200);
    for (uint32_t i = 0; i < MAX_CLIENTS; ++i) {
        g_evidence.after_200[i] = g_scheduler.clients[i].grants;
    }
    if (g_evidence.after_200[0] != 100) {
        return 4;
    }
    for (uint32_t i = 1; i < MAX_CLIENTS; ++i) {
        if (g_evidence.after_200[i] != 1) {
            return 5;
        }
    }
    g_evidence.stress_ok = 1;
    return 0;
}

__attribute__((noinline)) static int run_all(void)
{
    if (run_baseline() != 0) {
        return 10;
    }
    return run_stress();
}

int main(void)
{
    return run_all();
}
